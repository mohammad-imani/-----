## making AVR Calculator
در این قسمت میخواهیم با استفاده از keypad  و lcd که قبلا با انها کار کردیم استفاده کنیم و یک ماشین حساب ساده بسازیم تا در اینده به نتیجه نهایی این را به عنوان یک برنامه اضافه کنیم. 
<br><br>
در مرحله اول اولین کاری که باید انجام دهیم قرار دادن یک منو برای کاربر است که شامل دو گزینه است. یک راهنما برای کاربر و گزینه دیگر برای شروع به کار کردن. من قصد دارم هر منویی که نمایش داده میشود را در قالب یک تابع قرار دهم و در هر جا که نیاز بود ان را call کنم . البته این مشکلاتی را به همراه دارد . مثلا اینکه میتواند باعث overflow شود . که البته برای پروژه ای در این مقیاس قابل چشم پوشی است چون هنوز در مرحله ابتدایی و راه اندازی ان هستیم. در مراحل بعد به این مشکل هم رسیدگی خواهم کرد. در ابتدا قبل از اینکه بخواهیم وارد بخش اصلی و نوشتن توابع کاربر شویم . باید ماژول هل را مانند انچه که قبلا انجام میدادیم راه اندازی کنیم. 
```cpp
#include <Wire.h> 
#include <LiquidCrystal_I2C.h>
#include <Keypad.h>

const byte ROW = 4;
const byte COL = 3;

byte ROWS[ROW] = {2, 3, 4, 5}; 
byte COLS[COL] = {6, 7, 8};      

char keymap[ROW][COL] = {{'1','2','3'},
                         {'4','5','6'},
                         {'7','8','9'},
                         {'.','0','#'}};


Keypad pad(makeKeymap(keymap), ROWS, COLS, ROW, COL);
LiquidCrystal_I2C lcd(0x27,16,2);
```
در اینجا ابتدا کتابخانه های مورد نیاز را شامل کردیم سپس ابعدا و پین های مورد استفاده در کیپد و نقشه کلید را تعریف کردیم. سپس شی ها را از کلاس ها ساختیم. اگر به ساخت شی نمایشگر دفت کنید سه ورودی گرفته است که دوتای سمت راست ابعاد و موردی سمت چپ ادرس مربوط به ماژول مبدل I2C است که یک عدد در مبنای 16 یا HEX است. <br><br>
در مرحله بعد وارد تابع setup  میشویم و تنضیمات لازم را انجام میدهیم. شروع به کار کردن خشک و خالی زیاد جالب نیست . برای همین در این بخش چند خط کد برای هنگام روشن شدن دستگاه قرار میدهیم تا کار جذاب تر شود.
اما قبل از ان نیاز داریم تا یک کاراکتر جدید برای نمایشگر تعریف کنیم و ان هم یک کاراکتر به شکل مربع توپر است . در ابتدا باید ماتریس انرا تعریف کنیم و سپس کاراکتر را بسازیم.
```cpp
byte fullBlock[8] = {
  B11111, B11111, B11111, B11111,
  B11111, B11111, B11111, B11111
};
void setup()
{
                   
  lcd.init();
  // Print a message to the LCD.
  lcd.backlight();
  lcd.createChar(1,fullBlock);
  lcd.clear();
  for (int i = 0 ; i < 16 ; i++){
    lcd.setCursor(i,0);
    lcd.write(byte(1));
    lcd.setCursor(15-i,1);
    lcd.write(byte(1));
    delay(70);
    lcd.clear();
    }
   for (int i = 0 ; i < 16 ; i++){
    lcd.setCursor(15-i,0);
    lcd.write(byte(1));
    lcd.setCursor(i,1);
    lcd.write(byte(1));
    delay(70);
    lcd.clear();
    }
  lcd.clear();
  lcd.setCursor(1,0);
  lcd.print("AVR Calculator");
  lcd.setCursor(4,1);
  lcd.print("+ - * / ^");
  delay(1000);
  lcd.clear();
  
}
```
اکنون همه چیز اماده است تا وارد بخش نوشتن کد های مربوط به کار های اصلی شویم. در ابتدا باید تابع اول یه قرار است Welcome بنامیمش را بنویسیم. در این تابع همچنین منتظر ورودی میمانیم تا کاربر یکی از گزینه ها را انتخاب کند سپس ما عمل لازم را انجام دهیم.
```cpp
void Welcome(){
  lcd.clear();
  lcd.setCursor(1,0);
  lcd.print("AVR Calculator");
  lcd.setCursor(1,1);
  lcd.print("1)Guide 2)Start");
  key = pad.getKey();
  while (!key){key = pad.getKey();};
  if (key == '1'){ return Guide();};
  if (key == '2'){ return Start();};
  }
```
این بازگشتی بودن بسیار مضر است برای بردی مثل uno که دارای رم بسیار کمی است. به هر حال . در مرحله های بعدی باید توابع start و guide را بنویسیم .ولی قبل از همه این ها باید یک روش برای دریافت عدد از کاربر داشته باشیم. چون که ما اعداد را به صورد کاراکتری در یافت میکنیم و باید انها را به اعشاری تبدیل کنیم. شاید بپرسید که چرا نقشه کلید را اعشاری یا صحیح تعریف نکردم که دوباره به این کار نیاز نباشد. پاسخ این است که کاربر عدد را وقتی یکی یکی وارد میکند ما باید بتوانیم عدد جدید را به عدد قبلی بچسبانیم که این کار با استفاده از رشته ها راحت تر انجام میشود. همچنین در این توابع گاهی نیاز داریم که منتظر بمانیم تا کاربر ورودی وارد کند.
